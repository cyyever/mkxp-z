cmake_minimum_required(VERSION 3.22)
project(mkxp_offline_msl_tools LANGUAGES C CXX)
include(FetchContent)
# These tools are not linked into mkxp-z. Override FetchContent source dirs to
# reuse verified downloads; the normal download path checks exact archive hashes.
FetchContent_Declare(spirvcross
 URL https://api.github.com/repos/KhronosGroup/SPIRV-Cross/tarball/57e4efcc84112abcd426dc5ec248b5339bee04f2
 URL_HASH SHA256=5be4b0ee78036ed6358f5320c6070df5c5bd1cf05c6f55dc221795f1d4e7bd38)
FetchContent_Declare(shadercross
 URL https://api.github.com/repos/libsdl-org/SDL_shadercross/tarball/1ff05bec573988a98ef9e0260b4da44f512b8367
 URL_HASH SHA256=feb818c93c804cff0f9a1f3f0d0462bfff09f33bc8406da7f716e9cff018187c)
if(MKXP_SDL_SOURCE)
 set(SDL_SHARED OFF CACHE BOOL "" FORCE)
 set(SDL_STATIC ON CACHE BOOL "" FORCE)
 set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
 set(SDL_TESTS OFF CACHE BOOL "" FORCE)
 set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
 foreach(feature AUDIO VIDEO GPU RENDER CAMERA JOYSTICK HAPTIC HIDAPI SENSOR POWER DIALOG TRAY)
  set(SDL_${feature} OFF CACHE BOOL "" FORCE)
 endforeach()
 add_subdirectory("${MKXP_SDL_SOURCE}" sdl EXCLUDE_FROM_ALL)
else()
 find_package(SDL3 CONFIG REQUIRED COMPONENTS SDL3-static)
endif()
set(SPIRV_CROSS_CLI OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_SKIP_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(spirvcross)
add_library(spirv_cross_c ALIAS spirv-cross-c)
set(SDLSHADERCROSS_DXC OFF CACHE BOOL "" FORCE)
set(SDLSHADERCROSS_SHARED OFF CACHE BOOL "" FORCE)
set(SDLSHADERCROSS_STATIC ON CACHE BOOL "" FORCE)
set(SDLSHADERCROSS_SPIRVCROSS_SHARED OFF CACHE BOOL "" FORCE)
set(SDLSHADERCROSS_CLI_STATIC ON CACHE BOOL "" FORCE)
set(SDLSHADERCROSS_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(shadercross)
add_executable(mkxp-msl-entrypoint entrypoint.cpp)
target_compile_features(mkxp-msl-entrypoint PRIVATE cxx_std_20)
target_link_libraries(mkxp-msl-entrypoint PRIVATE spirv-cross-c)
if(MSVC)
 target_compile_options(mkxp-msl-entrypoint PRIVATE /W4 /WX /utf-8)
else()
 target_compile_options(mkxp-msl-entrypoint PRIVATE -Wall -Wextra -Werror)
endif()
